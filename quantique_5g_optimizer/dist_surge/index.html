Servo beamServo;
const int SERVO_PIN = 9;

void setup() {
  Serial.begin(115200);
  beamServo.attach(SERVO_PIN);
  beamServo.write(90); // Center at 0° Azimuth
}

void loop() {
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    if (cmd.startsWith("$HIL_TX")) {
      float az = parseAzimuth(cmd);
      beamServo.write(constrain(90.0 + az, 45, 135));
      animatePhaseLEDs(az);
      Serial.println("$HIL_ACK,DEV=PHASED_ARRAY,LOCKED=TRUE*0x98AE");
    }
  }
}
